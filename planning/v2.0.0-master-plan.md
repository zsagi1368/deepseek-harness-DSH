async function executeWithBudget(tool: ToolDefinition, args: unknown) {
  const active = getActiveToolCalls()
  if (tool.concurrencyLimit && active >= tool.concurrencyLimit) {
    await waitForSlot(tool.concurrencyLimit - active)
  }
  return tool.execute(args, exec)
}
